### 4.1 回吐減半與半倉對照

R1真正有減碼的窗口，A為2595／21194（12.244%），B為3169／29723（10.662%）。後段這些窗口平均少跌2.801點、少賺1.948點；全窗口差異正好等於觸發比例乘上觸發窗口差異。

|---|---:|---:|---:|---:|
| 全部ALL窗口 | 3.648 | 4.038 | 1.438 | 4.117 |

所以「C50在全部窗口的這個描述比值較高」成立；「一開始少買對所有狀態都更有效」就超出證據。子群排序反轉也不代表R1已可採納。兩種政策曝險量、時序都不同，還不能稱作風險相同下的公平優勝證明。

### 4.2 E1約四倍的算式，不等於加部位實驗

後段 `E1增加報酬0.825851點／增加回撤0.921022點 = 0.896668`；`B0平均報酬／平均回撤 = 0.235245`；再相除是3.811630。**算式重現，但前者是政策差分，後者是水準比值**。把後者當作增加B0部位的邊際交換率，需要另外假設回撤隨曝險線性變化。

百分比最大回撤不符合這個一般假設。例：資產由1升到1.20再跌到1.08，回撤10%；固定初始本金、部位兩倍，路徑成1、1.40、1.16，回撤17.14%，並非20%。若只是把本金及資產一起乘二，百分比回撤更是完全不變。

因此我同意「E1值得下一輪驗證」，但建議把「比直接加部位效率高約四倍」改為「兩個描述比值相除約3.81，尚未做同曝險或同風險反事實」。各股Calmar中位數也不等於一個共同股票組合的Calmar。

### 4.3 ATR及研究定位

ATR要包含昨收與今日高低價的跳空：`TR=max(H-L, abs(H-昨收), abs(L-昨收))`。本程式用Wilder ATR14，初始14個TR平均，之後 `(13×前ATR+本日TR)/14`；不能寫成只平均每天高低差。[Fidelity原始指標說明](https://www.fidelity.com/learning-center/trading-investing/technical-analysis/technical-indicator-guide/atr)

同一資料由另一份程式算對，是重要的工程複核，但不會多出一批獨立市場證據。E1原先已事前列為探索比較，現在看過結果後選作下一輪主候選；原結果仍須保留「探索」身分。換股仍與舊樣本共享市場時段，不足以自動解除選擇偏誤。[Bailey等的原始回測過擬合論文](https://www.davidhbailey.com/dhbpapers/backtest-prob.pdf)討論的正是反覆選規則與驗證的界線；本輪沒有計算該論文的PBO。

"""Economic OHLCV digests without download metadata; stdlib-only Python 3."""
import argparse, csv, hashlib, io, json, math
from pathlib import Path

def digest(value):
    return hashlib.sha256(json.dumps(value, ensure_ascii=True, separators=(',', ':'), allow_nan=False).encode()).hexdigest()

def normalize(bars):
    out=[]
    for bar in bars:
        if len(bar)!=6: raise ValueError('Expected [date,open,high,low,close,volume]')
        d,*v=bar
        if not isinstance(d,str) or len(d)!=10: raise ValueError('Invalid date')
        prices=[float(x) for x in v[:4]]
        if any(not math.isfinite(x) or x<=0 for x in prices): raise ValueError('Invalid price')
        volume=float(v[4])
        if not math.isfinite(volume) or volume<0 or not volume.is_integer(): raise ValueError('Invalid volume')
        out.append([d]+[x.hex() for x in prices]+[int(volume)])
    if any(out[i][0]>=out[i+1][0] for i in range(len(out)-1)): raise ValueError('Dates not strictly ascending')
    return out

def bar_hash(bars): return digest(normalize(bars))

def build(data_dir, verify=True, selected=None):
    manifest=json.loads((data_dir/'manifest.json').read_text())
    symbols={}
    for symbol,m in sorted(manifest['symbols'].items()):
        if selected and symbol not in selected: continue
        p=data_dir/m['adjustedFile']; blob=p.read_bytes()
        actual=hashlib.sha256(blob).hexdigest()
        if verify and actual!=m['adjustedSha256']: raise ValueError('File mismatch: '+symbol)
        x=json.loads(blob); bars=x['bars']
        if x['symbol']!=symbol: raise ValueError('Wrong symbol')
        symbols[symbol]={'bars':len(bars),'first':bars[0][0] if bars else None,'last':bars[-1][0] if bars else None,
                         'barsSha256':bar_hash(bars),'adjustedFileSha256':actual}
    cal=json.loads((data_dir/manifest['calendarFile']).read_text())
    return {'schema':'astra-economic-bars-v1','serialization':'JSON compact ASCII rows: date, float.hex OHLC, integer volume; metadata excluded',
            'manifestSha256':hashlib.sha256((data_dir/'manifest.json').read_bytes()).hexdigest(),
            'createdAt':manifest['createdAt'],'calendarDatesSha256':digest(cal['dates']),
            'calendarSessions':len(cal['dates']),'symbols':symbols}

def self_test():
    bars=[['2020-01-02',10,12,9,11,100]]
    a={'bars':bars,'retrievedAt':'a'}; b={'bars':bars,'retrievedAt':'b'}
    assert bar_hash(a['bars'])==bar_hash(b['bars'])
    assert bar_hash(bars)==bar_hash([['2020-01-02',10.0,12.,9.,11.,100.]])
    assert bar_hash(bars)!=bar_hash([['2020-01-02',10,12,9,11.00001,100]])
    assert bar_hash(bars)!=bar_hash([['2020-01-02',10,12,9,11,101]])
    try: normalize(bars+bars)
    except ValueError: pass
    else: raise AssertionError('Duplicate date accepted')
    return {'tests':5,'status':'PASS'}

def main():
    ap=argparse.ArgumentParser(description=__doc__)
    ap.add_argument('--data-dir',type=Path)
    ap.add_argument('--output',type=Path)
    ap.add_argument('--expect',type=Path,help='Astra original economic-bars.json to compare')
    ap.add_argument('--symbols',help='Optional comma-separated symbols')
    ap.add_argument('--self-test',action='store_true')
    args=ap.parse_args()
    if args.self_test: print(json.dumps(self_test())); return
    if args.data_dir is None or args.output is None: ap.error('--data-dir and --output required')
    out=build(args.data_dir, selected=set(args.symbols.split(',')) if args.symbols else None)
    args.output.parent.mkdir(parents=True,exist_ok=True)
    args.output.write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n')
    s=io.StringIO(newline=''); w=csv.writer(s,lineterminator='\n')
    w.writerow(['symbol','bars','first','last','barsSha256','adjustedFileSha256'])
    for symbol,row in out['symbols'].items(): w.writerow([symbol]+[row[k] for k in ['bars','first','last','barsSha256','adjustedFileSha256']])
    args.output.with_suffix('.csv').write_text(s.getvalue())
    print(json.dumps({'symbols':len(out['symbols']),'calendarSessions':out['calendarSessions'],'output':str(args.output)}))
    if args.expect:
        expected=json.loads(args.expect.read_text()); e=expected['symbols']; a=out['symbols']
        common=sorted(a.keys()&e.keys())
        comparison={'common':len(common),'onlyActual':sorted(a.keys()-e.keys()),'onlyExpected':sorted(e.keys()-a.keys()),
                    'calendarDatesEqual':out['calendarDatesSha256']==expected['calendarDatesSha256'],
                    'barsEqual':[s for s in common if a[s]['barsSha256']==e[s]['barsSha256']],
                    'barsDifferent':[s for s in common if a[s]['barsSha256']!=e[s]['barsSha256']],
                    'warning':'A difference is exact floating-value inequality, not a material-change diagnosis. Compare dates/values and trade events next.'}
        args.output.with_name(args.output.stem+'-comparison.json').write_text(json.dumps(comparison,indent=2)+'\n')
        print(json.dumps({k:v for k,v in comparison.items() if k not in ['barsEqual','warning']}))

if __name__=='__main__': main()

#!/usr/bin/env python3
"""Independent daily cash/share replay of the delivered Astra exit-study accounts.

No imports from experiment.py, data_tools.py, or previous audit algorithms.
Uses delivered episode dates/events as fixed instructions, NOT a new independent
implementation of entry or sell signal decisions. Prices, holdings, daily NAV,
MDD, compounding, calendar selection, CAGR and Calmar are recalculated here.
No network, strategy tuning, new tickers, or writes into the old study folder.
"""
from __future__ import annotations
import csv
import gzip
import hashlib
import json
import math
import statistics

from pathlib import Path

HERE = Path(__file__).resolve().parent
OLD = HERE.parent / 'astra-exits-20261013'
POLICIES = ('B0', 'R1', 'E1', 'C50')
PERIODS = {'A': ('2010-01-01', '2017-12-31'), 'B': ('2018-01-01', '2026-07-31')}
ATOL, RTOL = 1e-10, 1e-10


def sha(blob):
    return hashlib.sha256(blob).hexdigest()


def median(values):
    values = sorted(v for v in values if v is not None)
    if not values:
        return None
    k = len(values)//2
    return values[k] if len(values)%2 else (values[k-1]+values[k])/2


class Audit:
    def __init__(self):
        self.inputs = {}
        self.counts = defaultdict(int)
        self.failures = []
        self.max_differences = defaultdict(float)

    def read(self, path):
        path = Path(path)
        blob = path.read_bytes()
        self.inputs[str(path.relative_to(OLD))] = sha(blob)
        return blob

    def check(self, ok, label, detail=None):
        self.counts[label] += 1
        if not ok:
            self.failures.append({'check': label, 'detail': detail})

    def near(self, got, expected, label, detail=None):
        if got is None or expected is None:
            self.check(got is expected, label, detail)
            return
        diff = abs(got-expected)
        self.max_differences[label] = max(self.max_differences[label], diff)
        self.check(math.isfinite(got) and math.isfinite(expected) and
                   diff <= ATOL + RTOL*abs(expected), label,
                   {'where': detail, 'actual': got, 'expected': expected, 'difference': diff})


def raw_price_map(doc):
    """Rebuild provider-adjusted OHLC from the original downloaded Yahoo body."""
    r = doc['chart']['result'][0]
    quote = r['indicators']['quote'][0]
    adjusted_close = r['indicators']['adjclose'][0]['adjclose']
    zone = ZoneInfo(r['meta']['exchangeTimezoneName'])
    out = {}
    for i, timestamp in enumerate(r['timestamp']):
        day = datetime.fromtimestamp(timestamp, zone).date().isoformat()
        raw_close, ac = quote['close'][i], adjusted_close[i]
        if not all(isinstance(v, (int,float)) and math.isfinite(v) and v > 0 for v in (raw_close, ac)):
            continue
        factor = ac/raw_close
        vals = [quote[k][i] for k in ('open','high','low')]
        if not all(isinstance(v, (int,float)) and math.isfinite(v) and v > 0 for v in vals):
            continue
        out[day] = [v*factor for v in vals] + [ac]
    return out


def select_campaigns(rows, calendar, ci):
    """Fixed reset: allow a new close signal on the prior campaign end date."""
    chosen = []
    unlocked = -1
    for signal_day in sorted(rows):
        t = ci[signal_day]
        if t < unlocked:
            continue
        chosen.append(rows[signal_day])
        unlocked = t + 64
    return chosen


def replay(bars, selected, start, end, policy, audit, label):
    """Daily open transaction ledger, independent from episode NAV multiplication."""
    purchases = {}
    sales = defaultdict(list)
    window_ends = set()
    for ep in selected:
        buy_day = ep['entry_date']
        audit.check(buy_day not in purchases, 'unique_buy_day', label)
        purchases[buy_day] = ep['date']
        window_ends.add(ep['end_date'])
        fractions = 0.
        for day, fraction, reason in ep['events']:
            sales[day].append((ep['date'], fraction, reason))
            fractions += fraction
            audit.check(buy_day <= day <= ep['end_date'] and fraction > 0,
                        'sale_time_and_fraction', (label, ep['date'], day, fraction))
        audit.near(fractions, 1., 'campaign_full_liquidation', (label, ep['date']))

    cash = 1.
    shares = initial_shares = 0.
    active_signal = None
    peak = 1.
    peak_date = start
    maximum_drawdown = 0.
    worst_pair = {'peak_date': start, 'trough_date': start, 'peak_nav': 1., 'trough_nav': 1.}
    curve = [1.]
    dates = [d for d in bars if start <= d <= end]
    exposure_sum = 0.
    transactions = 0
    turnover = 0.
    for day in dates:
        opening, _, _, closing = bars[day]
        if day in purchases:
            audit.near(shares, 0., 'flat_before_new_purchase', (label, day))
            budget = cash*(.5 if policy == 'C50' else 1.)
            initial_shares = budget/opening
            shares = initial_shares
            cash -= budget
            active_signal = purchases[day]
            transactions += 1
            turnover += budget
        for signal_day, fraction, reason in sales.get(day, []):
            audit.check(signal_day == active_signal, 'sale_matches_current_campaign', (label, day))
            amount = initial_shares*fraction
            audit.check(amount <= shares+1e-10, 'no_oversell', (label, day))
            cash += amount*opening
            shares -= amount
            transactions += 1
            turnover += amount*opening
            if abs(shares) < 1e-12*max(1,initial_shares):
                shares = 0.
        audit.check(cash >= -1e-12 and shares >= -1e-12, 'no_borrowing', (label, day))
        if day in window_ends:
            audit.near(shares, 0., 'flat_on_common_end_date', (label, day))
        equity = cash + shares*closing
        curve.append(equity)
        exposure_sum += shares*closing/equity
        if equity > peak:
            peak, peak_date = equity, day
        dd = (peak-equity)/peak
        if dd > maximum_drawdown:
            maximum_drawdown = dd
            worst_pair = {'peak_date': peak_date, 'trough_date': day, 'peak_nav': peak, 'trough_nav': equity}
    audit.near(shares, 0., 'flat_at_account_end', label)
    years = (date.fromisoformat(end)-date.fromisoformat(start)).days/365.25
    annual = curve[-1]**(1/years)-1 if years > 0 and curve[-1] > 0 else None
    return {'return': curve[-1]-1, 'cagr': annual, 'mdd': maximum_drawdown,
            'calmar': annual/maximum_drawdown if maximum_drawdown and annual is not None else None,
            'exposure': exposure_sum/len(dates), 'transactions': transactions, 'turnover': turnover,
            'campaigns': len(selected), 'start': start, 'end': end, 'days': len(dates),
            'curve_points': len(curve), 'worst_drawdown': worst_pair,
            'selected_signal_dates': [ep['date'] for ep in selected], 'curve': curve}

                    for i,(actual,expected) in enumerate(zip(account['curve'],old_curve)):

## 核對範圍

沿用的部分只有已交付episode的合格買點日期、卖出日期及股數比例：**訊號策略決策不是本次獨立重寫**。固定重置t+64、次開盤進場、半本金C50、費用0與現金息0，皆由本程式重新處理。由原價完整日曆自行判斷200根暖機及全段完整性，再與舊帳戶納入名單逐筆比對。
